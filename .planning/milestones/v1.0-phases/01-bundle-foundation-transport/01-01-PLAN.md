    **1. Create `src/Validator/WebspaceLocaleValidator.php`**:
    ```php
    namespace Sulu\McpServerBundle\Validator;

    use Sulu\Component\Webspace\Manager\WebspaceManagerInterface;

    class WebspaceLocaleValidator
    {
        public function __construct(
            private readonly WebspaceManagerInterface $webspaceManager,
        ) {}

        /**
         * Validates that the given webspace exists and the locale is available in it.
         *
         * @throws \InvalidArgumentException if webspace or locale is invalid
         */
        public function validate(string $webspace, string $locale): void
        {
            $ws = $this->webspaceManager->findWebspaceByKey($webspace);
            if ($ws === null) {
                throw new \InvalidArgumentException(
                    sprintf("Unknown webspace: '%s'", $webspace)
                );
            }

            $locales = array_map(
                fn($l) => $l->getLocale(),
                $ws->getAllLocalizations()
            );

            if (!in_array($locale, $locales, true)) {
                throw new \InvalidArgumentException(
                    sprintf(
                        "Locale '%s' is not available in webspace '%s'. Available: %s",
                        $locale,
                        $webspace,
                        implode(', ', $locales)
                    )
                );
            }
        }
    }
    ```

    **2. Create `src/Tool/PingTool.php`** (per D-09, TRNS-03, LOCL-01):
    ```php
    namespace Sulu\McpServerBundle\Tool;

    use Mcp\Capability\Attribute\McpTool;
    use Sulu\McpServerBundle\Validator\WebspaceLocaleValidator;
    use Sulu\Component\Webspace\Manager\WebspaceManagerInterface;

    class PingTool
    {
        public function __construct(
            private readonly WebspaceLocaleValidator $validator,
            private readonly WebspaceManagerInterface $webspaceManager,
        ) {}

        #[McpTool(
            name: 'sulu_ping',
            description: 'Verify MCP connection and authentication. Returns server info and validates webspace/locale.',
        )]
        public function ping(string $webspace, string $locale): array
        {
            $this->validator->validate($webspace, $locale);

            $ws = $this->webspaceManager->findWebspaceByKey($webspace);
            $availableLocales = array_map(
                fn($l) => $l->getLocale(),
                $ws->getAllLocalizations()
            );

            return [
                'status' => 'ok',
                'server' => 'sulu-mcp-server',
                'version' => '1.0.0',
                'webspace' => $webspace,
                'locale' => $locale,
                'available_locales' => $availableLocales,
            ];
        }
    }
    ```
