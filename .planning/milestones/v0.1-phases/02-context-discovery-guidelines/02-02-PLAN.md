    **Step 3: src/Repository/ContentGuidelinesRepository.php**
    ```php
    <?php declare(strict_types=1);
    namespace Sulu\McpServerBundle\Repository;

    use Doctrine\Bundle\DoctrineBundle\Repository\ServiceEntityRepository;
    use Doctrine\Persistence\ManagerRegistry;
    use Sulu\McpServerBundle\Entity\ContentGuidelines;

    /**
     * @extends ServiceEntityRepository<ContentGuidelines>
     */
    class ContentGuidelinesRepository extends ServiceEntityRepository
    {
        public function __construct(ManagerRegistry $registry)
        {
            parent::__construct($registry, ContentGuidelines::class);
        }

        /**
         * Resolve guidelines for a webspace by merging global defaults with per-webspace overrides.
         * Pass 'global' or null to get global defaults only.
         *
         * @return array<string, mixed>
         */
        public function resolveForWebspace(string $webspaceKey): array
        {
            // Fetch global defaults (webspace = null)
            $global = $this->findOneBy(['webspace' => null]);
            $resolved = $this->toArray($global);

            // Return global defaults if requesting global
            if ('global' === $webspaceKey) {
                return $resolved;
            }

            // Merge per-webspace overrides: non-null webspace values replace global values
            $specific = $this->findOneBy(['webspace' => $webspaceKey]);
            if (null !== $specific) {
                foreach ($this->toArray($specific) as $field => $value) {
                    if (null !== $value && 'webspace' !== $field) {
                        $resolved[$field] = $value;
                    }
                }
            }

            return $resolved;
        }

        /** @return array<string, mixed> */
        private function toArray(?ContentGuidelines $entity): array
        {
            if (null === $entity) {
                return [
                    'webspace' => null,
                    'tone' => null,
                    'audience' => null,
                    'style' => null,
                    'brand_rules' => null,
                    'dos' => null,
                    "don'ts" => null,
                ];
            }
            return [
                'webspace' => $entity->getWebspace(),
                'tone' => $entity->getTone(),
                'audience' => $entity->getAudience(),
                'style' => $entity->getStyle(),
                'brand_rules' => $entity->getBrandRules(),
                'dos' => $entity->getDos(),
                "don'ts" => $entity->getDonts(),   // apostrophe in output key only
            ];
        }
    }
    ```

    **Step 7: src/Tool/UpdateGuidelinesTool.php**
    ```php
    <?php declare(strict_types=1);
    namespace Sulu\McpServerBundle\Tool;

    use Doctrine\ORM\EntityManagerInterface;
    use Mcp\Capability\Attribute\McpTool;
    use Sulu\McpServerBundle\Entity\ContentGuidelines;
    use Sulu\McpServerBundle\Repository\ContentGuidelinesRepository;

    class UpdateGuidelinesTool
    {
        public function __construct(
            private readonly ContentGuidelinesRepository $repository,
            private readonly EntityManagerInterface $entityManager,
        ) {}

        /**
         * @return array<string, mixed>
         */
        #[McpTool(
            name: 'sulu_update_guidelines',
            description: 'Create or update content guidelines for a webspace (or global defaults when webspace is null or "global"). Only provided fields are updated — omitted fields keep existing values. Total guidelines text should not exceed 2000 characters.',
        )]
        public function updateGuidelines(
            ?string $webspace = null,
            ?string $tone = null,
            ?string $audience = null,
            ?string $style = null,
            ?string $brandRules = null,
            ?string $dos = null,
            ?string $donts = null,
        ): array {
            // Normalize "global" keyword to null for DB storage
            $webspaceKey = ('global' === $webspace) ? null : $webspace;

            $entity = $this->repository->findOneBy(['webspace' => $webspaceKey])
                ?? new ContentGuidelines();

            $entity->setWebspace($webspaceKey);
            if (null !== $tone) { $entity->setTone($tone); }
            if (null !== $audience) { $entity->setAudience($audience); }
            if (null !== $style) { $entity->setStyle($style); }
            if (null !== $brandRules) { $entity->setBrandRules($brandRules); }
            if (null !== $dos) { $entity->setDos($dos); }
            if (null !== $donts) { $entity->setDonts($donts); }

            // Soft size warning (not a hard constraint)
            $totalLength = \strlen((string) $entity->getTone())
                + \strlen((string) $entity->getAudience())
                + \strlen((string) $entity->getStyle())
                + \strlen((string) $entity->getBrandRules())
                + \strlen((string) $entity->getDos())
                + \strlen((string) $entity->getDonts());

            $this->entityManager->persist($entity);
            $this->entityManager->flush();

            $result = [
                'success' => true,
                'webspace' => $webspaceKey ?? 'global',
            ];

            if ($totalLength > 2000) {
                $result['warning'] = \sprintf(
                    'Total guidelines text is %d characters. Recommended maximum is 2000 characters for optimal AI context usage.',
                    $totalLength
                );
            }

            return $result;
        }
    }
    ```
